## استفاده

```php
use Tavaan\Engine\Cache\CacheRepository;

$cache = new CacheRepository(['default' => 'array']);

// Basic
$cache->put('key', 'value', 3600);   // TTL ثانیه
$cache->get('key', 'default');
$cache->has('key');
$cache->forget('key');
$cache->flush();
$cache->forever('key', 'value');

// remember (with stampede protection)
$value = $cache->remember('expensive_key', 3600, fn() => computeExpensive());
$value = $cache->rememberForever('key', fn() => compute());

// add (only if missing)
$cache->add('key', 'value', 60);

// pull (get + forget)
$value = $cache->pull('key');

// increment / decrement
$cache->increment('counter');
$cache->increment('counter', 5);
$cache->decrement('counter');

// many
$values = $cache->many(['k1', 'k2']);
$cache->putMany(['k1' => 'v1', 'k2' => 'v2'], 60);

// store خاص
$cache->store('file')->put('key', 'value', 3600);

// Lock
$lock = $cache->store()->lock('order:12', seconds: 10);
if ($lock->acquire()) {
    try { /* critical section */ }
    finally { $lock->release(); }
}
// با block و timeout
$lock->block(5, fn() => /* ... */);
```
